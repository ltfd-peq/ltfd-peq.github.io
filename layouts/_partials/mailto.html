Uso: href="{{ partial "mailto.html" (dict "email" $x "assunto" "…") }}" */ -}} {{- $href := printf "mailto:%s" .email -}} {{- with .assunto }}{{ $href = printf "%s?subject=%s" $href (replace (urlquery .) "+" "%20") }}{{ end -}} {{- return ($href | safeURL) -}}
